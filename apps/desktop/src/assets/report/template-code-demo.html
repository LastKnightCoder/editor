function quickSort(arr, lo, hi) { if (lo >= hi) return; const pivot = partition(arr, lo, hi); quickSort(arr, lo, pivot - 1); quickSort(arr, pivot + 1, hi); } function partition(arr, lo, hi) { const pivot = arr[hi]; let i = lo - 1; for (let j = lo; j < hi; j++) { if (arr[j] <= pivot) { i++; [arr[i], arr[j]] = [arr[j], arr[i]]; } } [arr[i + 1], arr[hi]] = [arr[hi], arr[i + 1]]; return i + 1; } function randomPartition(arr, lo, hi) { const rand = lo + Math.floor(Math.random() * (hi - lo + 1)); [arr[rand], arr[hi]] = [arr[hi], arr[rand]]; return partition(arr, lo, hi); } function threeWayPartition(arr, lo, hi) { let lt = lo, gt = hi, i = lo; const pivot = arr[lo]; while (i <= gt) { if (arr[i] < pivot) swap(arr, lt++, i++); else if (arr[i] > pivot) swap(arr, i, gt--); else i++; } return [lt, gt]; }

快速排序（QuickSort）是由 Tony Hoare 于 1960 年提出的一种高效排序算法。它基于分治法的思想，通过选取一个pivot（基准值），将数组分为两个子数组——小于基准的元素和大于基准的元素，然后递归地对子数组进行排序。

本文将从基础的 Lomuto 分区方案出发，逐步优化到随机化选取和三路分区方案，最终达到工程级的实现质量。

Step 01

递归框架

快速排序的主函数采用经典的分治递归结构。接收数组及左右边界 lo 和 hi，通过 partition 函数将数组一分为二，然后对两个子数组递归排序。

递归终止条件为 lo >= hi，即子数组长度为 0 或 1 时自然有序。

Step 02

Lomuto 分区：初始化

Lomuto 分区方案选择数组最后一个元素作为基准值（pivot）。变量 i 维护"小于等于 pivot 区域"的右边界。

这种方案实现简单直观，但在已排序数组上会退化为 O(n²)。

Step 03

Lomuto 分区：扫描与交换

指针 j 从左向右扫描。遇到小于等于 pivot 的元素时，将 i 右移并与 j 位置交换。这确保了 [lo..i] 区间内所有元素都不大于 pivot。

循环不变量：arr[lo..i] <= pivot，arr[i+1..j-1] > pivot。

Step 04

Lomuto 分区：放置基准

扫描完成后，将 pivot 从末尾交换到 i+1 位置。此时 pivot 左侧所有元素 <= pivot，右侧所有元素 > pivot。分区完成，返回 pivot 的最终索引。

Step 05

优化：随机化 Pivot

基础版本在有序输入上性能极差。通过随机选取 pivot 并与末尾交换，可以将最坏情况的概率降至极低。期望时间复杂度始终为 O(n log n)。

这是工程实现中最常用的优化手段之一。

Step 06

优化：三路分区

当数组中存在大量重复元素时，二路分区会产生不平衡的划分。三路分区（荷兰国旗问题）将数组分为三段：< pivot、== pivot、> pivot。

三个指针 lt、i、gt 协同工作，只对 < pivot 和 > pivot 的部分递归，等于 pivot 的部分已在正确位置。这在含大量重复元素的数组上性能显著提升。

复杂度分析

时间与空间复杂度全景

理解快速排序的复杂度特征是正确使用它的前提。与归并排序不同，快速排序的性能高度依赖基准值（pivot）的选取质量。一个好的 pivot 能将数组均匀分割，而最差的 pivot 则让递归树退化为线性链表。

优化策略

工程级优化技巧

在实际工程中，朴素的快速排序无法满足生产环境的性能要求。现代排序库通常会组合多种优化策略，将快速排序的常数因子降到最低，同时规避最坏情况。以下是四种关键优化手段。

Strategy 01
小数组插入排序
当子数组长度小于阈值（通常 8-16）时，切换为插入排序。插入排序在小规模数据上因低开销和缓存友好性表现优于递归式快排。
const THRESHOLD = 16;
function hybridSort(arr, lo, hi) {
if (hi - lo < THRESHOLD) {
insertionSort(arr, lo, hi);
return;
}
// ... quicksort logic
}
Strategy 02
三数取中法
从子数组的首、中、末三个元素中选取中位数作为 pivot。这能有效避免已排序输入的退化，且开销极小（仅 3 次比较）。
function medianOfThree(arr, lo, hi) {
const mid = (lo + hi) >>> 1;
if (arr[mid] < arr[lo])
swap(arr, lo, mid);
if (arr[hi] < arr[lo])
swap(arr, lo, hi);
if (arr[mid] < arr[hi])
swap(arr, mid, hi);
return arr[hi]; // median as pivot
}
Strategy 03
尾递归消除
通过将较大的子数组用循环处理（仅递归较小的那一半），将最坏情况的栈深度从 O(n) 降至 O(log n)，防止栈溢出。
function quickSortTailOpt(arr, lo, hi) {
while (lo < hi) {
const p = partition(arr, lo, hi);
// 递归较小的分区
if (p - lo < hi - p) {
quickSortTailOpt(arr, lo, p - 1);
lo = p + 1; // 尾递归优化
} else {
quickSortTailOpt(arr, p + 1, hi);
hi = p - 1;
}
}
}
Strategy 04
并行快速排序
利用 Web Workers 或多线程将两个子数组的递归排序并行化。当数据量超过百万级时，可获得接近线性的加速比。
async function parallelSort(arr, lo, hi, depth) {
if (hi - lo < 10000 || depth > 3) {
quickSort(arr, lo, hi); return;
}
const p = partition(arr, lo, hi);
await Promise.all([
parallelSort(arr, lo, p-1, depth+1),
parallelSort(arr, p+1, hi, depth+1),
]);
}
实际应用

主流语言的排序实现

现代编程语言的标准库排序函数几乎都不是"纯"快速排序，而是结合了多种算法的混合排序。它们吸取了快速排序的高性能特征，同时通过回退机制规避其最坏情况。理解这些实现有助于我们在算法选择和性能调优中做出更明智的决策。

JavaScript / V8
TimSort
V8 引擎（Chrome/Node.js）自 2019 年起使用 TimSort 替代了原有的快速排序实现。TimSort 是一种稳定的混合排序，结合了归并排序和插入排序，擅长处理部分有序数据。
// V8 的 Array.prototype.sort 使用 TimSort
const arr = [3, 1, 4, 1, 5, 9, 2, 6];
// 稳定排序，保证相等元素的相对顺序
arr.sort((a, b) => a - b);
// 内部流程：
// 1. 找到自然有序的 "run"
// 2. 短 run 用插入排序扩展到 minRun
// 3. 用归并合并相邻 run
Python
TimSort (原创)
Python 是 TimSort 的发源地（2002 年由 Tim Peters 设计）。CPython 的 list.sort() 和 sorted() 均使用此算法。它对现实世界中常见的"几乎有序"数据表现极佳。
# Python 的排序保证稳定且自适应
data = [('Alice', 85), ('Bob', 92), ('Carol', 85)]
# key 参数 + 稳定排序 = 多键排序
result = sorted(data, key=lambda x: x[1], reverse=True)
# 对已排序数据：O(n) 时间
# 对随机数据：O(n log n) 时间
# 最坏情况：O(n log n) 时间（有保证）
Rust
pdqsort
Rust 标准库的不稳定排序使用 pdqsort（Pattern-defeating Quicksort），由 Orson Peters 设计。它是 Introsort 的现代改进版，能自动检测输入模式并选择最优策略。
// Rust 的 sort_unstable 使用 pdqsort
let mut vec = vec![5, 2, 8, 1, 9, 3];
vec.sort_unstable();
// pdqsort 的策略选择：
// - 小数组 -> 插入排序
// - 检测到不良 pivot -> 切换到 heapsort
// - 检测到已排序 -> 线性扫描确认
// - 大量重复 -> 三路分区
// 最坏情况保证 O(n log n)
◇ 工程启示

现代排序实现的核心设计哲学是"自适应"——算法会在运行时检测输入数据的特征，动态切换策略。这启示我们：在实际项目中，不要追求单一算法的极致优化，而应该根据数据规模和分布特征选择合适的策略组合。

关键对比：TimSort 是稳定排序（保证相等元素的顺序），适合需要多键排序的场景；pdqsort 是不稳定排序，但常数因子更小，适合对性能要求极致的场景。JavaScript 在 ECMAScript 2019 后规范要求 Array.sort 必须稳定。

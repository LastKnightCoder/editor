```typescript
import { RenderElementProps } from "slate-react";
import Base from "../base";
import IExtension from "../types";
import type { ChartElement } from "../../types";
import Chart from "./components/Chart";
import blockPanelItems from "./block-panel-items";
import hotkeys from "./hotkeys";
import { createBlockElementPlugin, createVoidElementPlugin } from "../../utils";

class ChartExtension extends Base implements IExtension {
  type = "chart";

  override getPlugins() {
    return [
      createBlockElementPlugin(this.type),
      createVoidElementPlugin(this.type),
    ];
  }

  override getBlockPanelItems() {
    return blockPanelItems;
  }

  override getHotkeyConfigs() {
    return hotkeys;
  }

  render(props: RenderElementProps) {
    const { element, attributes, children } = props;
    return (
      <Chart element={element as ChartElement} attributes={attributes}>
        {children}
      </Chart>
    );
  }
}

export default ChartExtension;
```

### 步骤 4: 创建 React 组件

**文件**: `packages/editor/src/extensions/chart/components/Chart/index.tsx`

```typescript
import React, { useMemo } from "react";
import { RenderElementProps } from "slate-react";
import { useSlateStatic, useReadOnly, ReactEditor } from "slate-react";
import { Transforms } from "slate";
import { IExtensionBaseProps } from "../../../../extensions/types";
import type { ChartElement } from "../../../../types";
import AddParagraph from "../../../../components/AddParagraph";
import {
  LineChart,
  Line,
  BarChart,
  Bar,
  PieChart,
  Pie,
  XAxis,
  YAxis,
  CartesianGrid,
  Tooltip,
  Legend,
  ResponsiveContainer,
} from "recharts";
import styles from "./index.module.less";

const Chart: React.FC<IExtensionBaseProps<ChartElement>> = (props) => {
  const { attributes, element, children } = props;
  const editor = useSlateStatic();
  const readOnly = useReadOnly();

  // 渲染图表
  const renderChart = () => {
    const { chartType, data, config } = element;
    const { width = 600, height = 400, colors = ["#8884d8"] } = config || {};

    switch (chartType) {
      case "line":
        return (
          <ResponsiveContainer width={width} height={height}>
            <LineChart data={data}>
              <CartesianGrid strokeDasharray="3 3" />
              <XAxis dataKey="label" />
              <YAxis />
              <Tooltip />
              <Legend />
              <Line type="monotone" dataKey="value" stroke={colors[0]} />
            </LineChart>
          </ResponsiveContainer>
        );

      case "bar":
        return (
          <ResponsiveContainer width={width} height={height}>
            <BarChart data={data}>
              <CartesianGrid strokeDasharray="3 3" />
              <XAxis dataKey="label" />
              <YAxis />
              <Tooltip />
              <Legend />
              <Bar dataKey="value" fill={colors[0]} />
            </BarChart>
          </ResponsiveContainer>
        );

      case "pie":
        return (
          <ResponsiveContainer width={width} height={height}>
            <PieChart>
              <Pie
                data={data}
                dataKey="value"
                nameKey="label"
                cx="50%"
                cy="50%"
                outerRadius={100}
                fill={colors[0]}
                label
              />
              <Tooltip />
            </PieChart>
          </ResponsiveContainer>
        );

      default:
        return <div>不支持的图表类型</div>;
    }
  };

  // 编辑数据
  const handleEditData = () => {
    const newData = prompt("输入 JSON 格式的数据:");
    if (newData) {
      try {
        const parsedData = JSON.parse(newData);
        const path = ReactEditor.findPath(editor, element);
        Transforms.setNodes(editor, { data: parsedData }, { at: path });
      } catch (e) {
        alert("JSON 格式错误");
      }
    }
  };

  return (
    <div {...attributes} contentEditable={false} className={styles.chartWrapper}>
      <div className={styles.chartContainer}>
        {element.data.length > 0 ? (
          renderChart()
        ) : (
          <div className={styles.placeholder}>
            暂无数据，点击编辑按钮添加数据
          </div>
        )}
      </div>

      {!readOnly && (
        <div className={styles.toolbar}>
          <button onClick={handleEditData}>编辑数据</button>
          <button
            onClick={() => {
              const path = ReactEditor.findPath(editor, element);
              Transforms.setNodes(editor, { chartType: "line" }, { at: path });
            }}
          >
            折线图
          </button>
          <button
            onClick={() => {
              const path = ReactEditor.findPath(editor, element);
              Transforms.setNodes(editor, { chartType: "bar" }, { at: path });
            }}
          >
            柱状图
          </button>
          <button
            onClick={() => {
              const path = ReactEditor.findPath(editor, element);
              Transforms.setNodes(editor, { chartType: "pie" }, { at: path });
            }}
          >
            饼图
          </button>
        </div>
      )}

      {children}
      <AddParagraph element={element} />
    </div>
  );
};

export default Chart;
```

### 步骤 5: 实现插件 (可选)

**文件**: `packages/editor/src/extensions/chart/plugins/index.ts`

```typescript
import { Editor } from "slate";

export const pasteChart = (editor: Editor) => {
  const { insertData } = editor;

  editor.insertData = (data: DataTransfer) => {
    const text = data.getData("text/plain");

    // 尝试解析为 JSON
    try {
      const parsed = JSON.parse(text);
      if (
        Array.isArray(parsed) &&
        parsed.every((item) => item.label && item.value)
      ) {
        // 插入图表
        const { insertChart } = require("../../../utils/insertElement");
        insertChart(editor, {
          chartType: "bar",
          data: parsed,
        });
        return;
      }
    } catch (e) {
      // 不是图表数据，继续默认处理
    }

    insertData(data);
  };

  return editor;
};
```

## 4.3 最佳实践和注意事项

### 必须遵守的规则

**1. 属性传递**:

```typescript
// ✅ 正确：必须传递 attributes 到根 DOM 元素
<div {...attributes} contentEditable={false}>
  {/* 内容 */}
  {children}
</div>

// ❌ 错误：没有传递 attributes
<div contentEditable={false}>
  {/* 内容 */}
  {children}
</div>
```

**2. 渲染 children**:

```typescript
// ✅ 正确：即使是 void 元素也必须渲染 children
<div {...attributes} contentEditable={false}>
  {/* 内容 */}
  {children}
</div>

// ❌ 错误：没有渲染 children
<div {...attributes} contentEditable={false}>
  {/* 内容 */}
</div>
```

**3. 可编辑性控制**:

```typescript
// Void 元素必须设置 contentEditable={false}
<div {...attributes} contentEditable={false}>
  {/* 内容不可编辑 */}
  {children}
</div>

// 容器元素不需要设置 (默认继承)
<blockquote {...attributes}>
  {/* 内容可编辑 */}
  {children}
</blockquote>
```

**4. 插件顺序**:

```typescript
override getPlugins() {
  return [
    // 自定义行为插件在前
    pasteHandler,
    markdownSyntax,
    resourceManagement,

    // Block/Inline/Void 插件在后
    createBlockElementPlugin(this.type),
    createVoidElementPlugin(this.type),
  ];
}
```

**5. 类型安全**:

```typescript
// ✅ 正确：在 types/element/ 中定义类型
// types/element/chart.ts
export interface ChartElement {
  type: "chart";
  // ...
}

// ✅ 正确：添加到联合类型
// types/custom-element.ts
export type CustomElement = ... | ChartElement;

// ✅ 正确：在 render 中进行类型断言
render(props: RenderElementProps) {
  return <Chart element={props.element as ChartElement} {...props} />;
}
```

### 性能优化建议

**1. 使用 React.memo**:

```typescript
const MemoizedChart = React.memo(Chart);

render(props: RenderElementProps) {
  return <MemoizedChart {...props} />;
}
```

**2. 避免在 render 中创建新对象**:

```typescript
// ❌ 错误：每次渲染都创建新对象
render(props) {
  const style = { padding: "16px" };
  return <div style={style}>{...}</div>;
}

// ✅ 正确：使用常量或 useMemo
const CHART_STYLE = { padding: "16px" };

render(props) {
  return <div style={CHART_STYLE}>{...}</div>;
}
```

**3. 使用 useMemoizedFn**:

```typescript
import { useMemoizedFn } from "ahooks";

const Chart = (props) => {
  // ✅ 正确：使用 useMemoizedFn 缓存回调
  const handleEditData = useMemoizedFn(() => {
    // ...
  });

  return <button onClick={handleEditData}>编辑</button>;
};
```

### 常见陷阱

**1. 忘记渲染 children**:

```typescript
// ❌ 错误：会导致 Slate 报错 "Cannot resolve a DOM node from Slate node"
<div {...attributes}>{/* 忘记渲染 children */}</div>

// ✅ 正确
<div {...attributes}>{children}</div>
```

**2. 忘记传递 attributes**:

```typescript
// ❌ 错误：会破坏编辑器功能 (选区、拖拽等)
<div>{children}</div>

// ✅ 正确
<div {...attributes}>{children}</div>
```

**3. 在插件中直接修改 state**:

```typescript
// ❌ 错误：直接修改节点
export const myPlugin = (editor: Editor) => {
  editor.insertText = (text) => {
    const node = getCurrentNode(editor);
    node.text += text; // 错误！不可变数据模型
  };
};

// ✅ 正确：使用 Transforms API
export const myPlugin = (editor: Editor) => {
  editor.insertText = (text) => {
    const path = getCurrentPath(editor);
    Transforms.insertText(editor, text, { at: path });
  };
};
```

**4. 忘记添加到类型系统**:

```typescript
// ❌ 错误：TypeScript 会报错
// CustomElement 联合类型中没有 ChartElement

// ✅ 正确：添加到联合类型
export type CustomElement = ... | ChartElement;
```

**5. Void 元素没有设置 contentEditable={false}**:

```typescript
// ❌ 错误：用户可以编辑内容，导致数据不一致
<div {...attributes}>
  <img src={element.src} />
  {children}
</div>

// ✅ 正确：设置 contentEditable={false}
<div {...attributes} contentEditable={false}>
  <img src={element.src} />
  {children}
</div>
```

### 推荐的插件功能

**1. Markdown 语法支持**:

```typescript
export const markdownSyntax = (editor: Editor) => {
  const { insertBreak } = editor;

  editor.insertBreak = () => {
    const currentNode = getCurrentTextNode(editor);
    if (!currentNode) {
      insertBreak();
      return;
    }

    const [node, path] = currentNode;
    const parentPath = Path.parent(path);
    const [parentNode] = Editor.node(editor, parentPath);

    // 检测特定模式并转换
    if (parentNode.type === "paragraph") {
      const text = Node.string(parentNode);
      if (text.match(/^::chart\s*$/)) {
        // 删除段落
        Transforms.removeNodes(editor, { at: parentPath });
        // 插入图表
        insertChart(editor, { chartType: "line", data: [] });
        return;
      }
    }

    insertBreak();
  };

  return editor;
};
```

**2. 粘贴处理**:

```typescript
export const pasteHandler = (editor: Editor) => {
  const { insertData } = editor;

  editor.insertData = async (data: DataTransfer) => {
    const { files } = data;

    // 处理特定 MIME 类型
    if (files && files.length > 0) {
      const file = files[0];
      if (file.type === "application/json") {
        try {
          const text = await file.text();
          const parsed = JSON.parse(text);
          if (isChartData(parsed)) {
            insertChart(editor, { chartType: "bar", data: parsed });
            return;
          }
        } catch (e) {
          // 继续默认处理
        }
      }
    }

    insertData(data);
  };

  return editor;
};
```

**3. 资源管理**:

```typescript
export const resourceManagement = (editor: Editor) => {
  const { apply } = editor;

  editor.apply = (op) => {
    // 删除节点时清理资源
    if (op.type === "remove_node" && op.node.type === "chart") {
      const chartElement = op.node as ChartElement;
      // 清理相关资源
      editor.services?.resource.decrementRef(chartElement.id);
    }

    // 插入节点时创建资源引用
    if (op.type === "insert_node" && op.node.type === "chart") {
      const chartElement = op.node as ChartElement;
      editor.services?.resource.create({
        id: chartElement.id,
        type: "chart",
        data: chartElement.data,
      });
    }

    apply(op);
  };

  return editor;
};
```

**4. 规范化处理**:

```typescript
export const normalize = (editor: Editor) => {
  const { normalizeNode } = editor;

  editor.normalizeNode = (entry) => {
    const [node, path] = entry;

    if (node.type === "chart") {
      const chartElement = node as ChartElement;

      // 确保有默认数据
      if (!chartElement.data || chartElement.data.length === 0) {
        Transforms.setNodes(
          editor,
          {
            data: [
              { label: "示例 A", value: 10 },
              { label: "示例 B", value: 20 },
            ],
          },
          { at: path },
        );
        return;
      }

      // 确保 children 有默认值
      if (!chartElement.children || chartElement.children.length === 0) {
        Transforms.insertNodes(
          editor,
          { type: "formatted", text: "" },
          { at: [...path, 0] },
        );
        return;
      }
    }

    normalizeNode(entry);
  };

  return editor;
};
```

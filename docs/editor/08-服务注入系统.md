## 8.2 资源管理服务

**ResourceService 接口**:

```typescript
export interface ResourceService {
  // 创建资源引用
  create: (resource: {
    src: string;
    type: "image" | "audio" | "video" | "whiteboard";
    data?: any;
  }) => void;

  // 递增引用计数
  incrementRef: (src: string) => void;

  // 递减引用计数
  decrementRef: (src: string) => void;
}
```

**使用示例**:

```typescript
// 在插件中管理资源生命周期
export const overwrite = (editor: Editor) => {
  const { apply } = editor;

  editor.apply = (op) => {
    // 删除音频时，释放资源引用
    if (op.type === "remove_node" && op.node.type === "audio") {
      const audioElement = op.node as AudioElement;
      if (audioElement.src && !audioElement.uploading) {
        editor.services?.resource.decrementRef(audioElement.src);
      }
    }

    // 插入音频时，创建资源引用
    if (op.type === "insert_node" && op.node.type === "audio") {
      const audioElement = op.node as AudioElement;
      if (audioElement.src && !audioElement.uploading) {
        editor.services?.resource.create({
          src: audioElement.src,
          type: "audio",
        });
      }
    }

    apply(op);
  };

  return editor;
};
```

## 8.3 文件服务

**FileService 接口**:

```typescript
export interface FileService {
  // 上传文件
  uploadFile: (file: File) => Promise<{
    src: string;
    name: string;
    size: number;
  }>;

  // 选择文件
  selectFile: (options: {
    accept?: string;
    multiple?: boolean;
  }) => Promise<File[]>;
}
```

**使用示例**:

```typescript
// 在组件中上传文件
const handleUpload = async (file: File) => {
  const editor = useSlateStatic();

  try {
    const resource = await editor.services?.file.uploadFile(file);
    if (resource) {
      const path = ReactEditor.findPath(editor, element);
      Transforms.setNodes(
        editor,
        { src: resource.src, uploading: false },
        { at: path },
      );
    }
  } catch (error) {
    console.error("Upload failed:", error);
  }
};
```

## 8.4 AI 服务

**AIService 接口**:

```typescript
export interface AIService {
  // 流式对话
  streamChat: (params: {
    messages: Message[];
    onChunk: (chunk: string) => void;
    onEnd: () => void;
    onError: (error: Error) => void;
  }) => void;

  // 文本转语音
  textToSpeech: (params: { text: string; voice?: string }) => Promise<{
    src: string;
  }>;
}
```

**使用示例**:

```typescript
// 在 AI 组件中使用流式对话
const handleGenerate = () => {
  const editor = useSlateStatic();
  const path = ReactEditor.findPath(editor, element);

  editor.services?.ai.streamChat({
    messages: [{ role: "user", content: prompt }],
    onChunk: (chunk) => {
      // 更新内容
      Transforms.setNodes(
        editor,
        { content: element.content + chunk },
        { at: path },
      );
    },
    onEnd: () => {
      // 完成
      Transforms.setNodes(editor, { generating: false }, { at: path });
    },
    onError: (error) => {
      console.error("AI generation failed:", error);
    },
  });
};
```

## 8.5 使用方式

### 注入服务

**在应用层注入**:

```typescript
import { EditorServicesProvider } from "@tau/editor";

const App = () => {
  const services: EditorServices = {
    resource: myResourceService,
    file: myFileService,
    media: myMediaService,
    ai: myAIService,
    compiler: myCompilerService,
    utility: myUtilityService,
    // ... 其他服务
  };

  return (
    <EditorServicesProvider services={services}>
      <Editor />
    </EditorServicesProvider>
  );
};
```

### 使用服务

**在插件中**:

```typescript
export const myPlugin = (editor: Editor) => {
  editor.insertData = async (data: DataTransfer) => {
    const file = data.files[0];

    // 使用服务上传文件
    const resource = await editor.services?.file.uploadFile(file);

    // ...
  };

  return editor;
};
```

**在组件中**:

```typescript
import { useSlateStatic } from "slate-react";

const MyComponent = (props) => {
  const editor = useSlateStatic();

  const handleAction = async () => {
    // 使用服务
    const result = await editor.services?.someService.someMethod();
    // ...
  };

  return <button onClick={handleAction}>操作</button>;
};
```

## 8.6 外部依赖解耦

通过服务注入系统，编辑器核心不依赖具体实现，保持纯净和可测试。

**优势**:

- 编辑器核心代码独立，不依赖外部库
- 应用层可以自由选择实现方式
- 易于测试和 mock
- 支持多种部署环境

**示例**:

```typescript
// 编辑器核心只定义接口
export interface FileService {
  uploadFile: (file: File) => Promise<{...}>;
}

// 应用层提供具体实现
const myFileService: FileService = {
  uploadFile: async (file) => {
    // 使用自己的上传逻辑（S3、OSS、本地等）
    return await uploadToMyServer(file);
  },
};
```

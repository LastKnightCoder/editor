
**文件位置**: `packages/editor/src/utils/insertElement.ts` (940 行)

### 核心函数

**setOrInsertNode - 智能插入**:

```typescript
export const setOrInsertNode = (
  editor: Editor,
  node: BlockElement,
  options: InsertElementOptions = {}
): Path | undefined
```

**功能**:

- 如果当前段落为空，替换为新节点
- 否则，在当前位置后插入新节点
- 返回插入的路径

**replaceNode - 替换节点**:

```typescript
export const replaceNode = (
  editor: Editor,
  node: BlockElement,
  match: (n: Node) => boolean,
  options: InsertElementOptions = {}
): Path | undefined
```

**功能**:

- 查找匹配的节点并替换

## 9.3 块导航

**文件位置**: `packages/editor/src/utils/blockNavigation.ts`

### 接口定义

```typescript
export interface BlockNavigationMeta {
  reason: "delete" | "arrow" | "enter"; // 触发原因
  direction: "forward" | "backward"; // 导航方向
}

export interface BlockFocusContext {
  editor: Editor;
  element: SlateElement;
  meta: BlockNavigationMeta;
}

export interface BlockNavigationHandler {
  focusStart?: (context: BlockFocusContext) => boolean;
  focusEnd?: (context: BlockFocusContext) => boolean;
}
```

### 注册和使用

```typescript
// 注册块导航处理器
registerBlockNavigation(
  element: SlateElement,
  handler: BlockNavigationHandler
): void

// 注销块导航处理器
unregisterBlockNavigation(element: SlateElement): void

// 聚焦元素
focusElementByNavigator(
  editor: Editor,
  element: SlateElement,
  position: "start" | "end",
  meta: BlockNavigationMeta
): boolean

// 获取兄弟块
getSiblingBlockEntry(
  editor: Editor,
  path: Path,
  direction: "forward" | "backward"
): [SlateElement, Path] | null
```

### 使用示例

```typescript
const CodeBlock = ({ element }) => {
  const codeMirrorRef = useRef<CodeMirrorEditor | null>(null);

  useEffect(() => {
    if (!codeMirrorRef.current) return;

    const handler: BlockNavigationHandler = {
      focusStart: ({ editor, element, meta }) => {
        const cm = codeMirrorRef.current;
        if (!cm) return false;

        cm.focus();
        cm.setCursor(0, 0);
        return true;
      },

      focusEnd: ({ editor, element, meta }) => {
        const cm = codeMirrorRef.current;
        if (!cm) return false;

        cm.focus();
        const lastLine = cm.lastLine();
        const lastLineLength = cm.getLine(lastLine).length;
        cm.setCursor(lastLine, lastLineLength);
        return true;
      },
    };

    registerBlockNavigation(element, handler);

    return () => {
      unregisterBlockNavigation(element);
    };
  }, [element]);

  // ...
};
```

## 9.4 编辑器操作

**文件位置**: `packages/editor/src/utils/editor.ts`

### 常用函数

```typescript
// 判断选区是否折叠
isCollapsed(editor: Editor): boolean

// 获取当前文本节点
getCurrentTextNode(editor: Editor): [Text, Path] | null

// 判断段落是否为空
isParagraphAndEmpty(editor: Editor): boolean

// 获取最近的块级元素
getClosestCurrentElement(editor: Editor): [CustomElement, Path] | null
```

### 使用示例

```typescript
import { isCollapsed, getClosestCurrentElement } from "./utils/editor";

// 在插件中使用
export const myPlugin = (editor: Editor) => {
  editor.insertBreak = () => {
    // 检查选区是否折叠
    if (!isCollapsed(editor)) {
      insertBreak();
      return;
    }

    // 获取最近的块级元素
    const current = getClosestCurrentElement(editor);
    if (current) {
      const [element, path] = current;

      // 根据元素类型执行不同操作
      if (element.type === "code-block") {
        // 代码块中的回车处理
        insertCodeBlockNewLine(editor);
        return;
      }
    }

    insertBreak();
  };

  return editor;
};
```
